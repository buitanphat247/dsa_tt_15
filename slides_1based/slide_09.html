Thuật Toán Vận Hành Của Trie

struct TrieNode {
    std::unordered_map children;
    bool is_end_of_word = false;
};

// 1. Thao tác Chèn từ: O(L)
void insert(TrieNode* root, const string& word) {
    TrieNode* curr = root;
    for (char ch : word) {
        if (!curr->children.count(ch))
            curr->children[ch] = new TrieNode();
        curr = curr->children[ch];
    }
    curr->is_end_of_word = true; // Bật cờ kết thúc từ
}

// 2. Thao tác Autocomplete (Prefix Query): O(L + K)
void collectKeys(TrieNode* curr, string pref, vector& res) {
    if (curr->is_end_of_word) res.push_back(pref);
    for (auto& [ch, child] : curr->children) {
        collectKeys(child, pref + ch, res); // Duyệt sâu DFS
    }
}
Trái tim của thuật toán Autocomplete nằm ở hai bước: Đầu tiên đi bộ L bước để đứng tại nút tiền tố trong O(L). Sau đó chạy DFS xuống cây con phía dưới để gom tất cả các từ có cờ is_end = true. Mọi thứ diễn ra mượt mà và cực kỳ trực quan.
